تست سریع جابه‌جایی بین دو حساب کاربری مختلف (شبیه‌سازی دو دستگاه):

برای ذخیره دائمی در دیتابیس ابری واقعی بدون نیاز به سرور شخصی، آدرس پروژه و کلید رایگان Supabase (PostgreSQL) خود را وارد کنید تا تمام تراکنش‌ها با جدول marz_ledgers همگام شوند: 

CREATE TABLE IF NOT EXISTS public.marz_ledgers (
  user_email TEXT PRIMARY KEY,
  payload JSONB NOT NULL,
  updated_at TIMESTAMPTZ DEFAULT now()
);
ALTER TABLE public.marz_ledgers ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public ledger access" ON public.marz_ledgers
  FOR ALL USING (true) WITH CHECK (true);
